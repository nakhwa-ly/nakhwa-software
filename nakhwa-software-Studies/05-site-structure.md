## 🔗 الروابط الداخلية الذكية

استخدم دائماً `Link` من `@/i18n/routing` وليس `next/link`:

```tsx
// ✅ صحيح
import { Link } from '@/i18n/routing';

<Link href="/services">خدماتنا</Link>
// يصبح تلقائياً /ar/services أو /en/services

// ❌ خطأ
import Link from 'next/link';
<Link href="/services">  // سيكسر الـ locale
```

## 📊 Metadata لكل صفحة

### مثال للصفحة الرئيسية
```tsx
import { getTranslations } from 'next-intl/server';

export async function generateMetadata({
  params,
}: {
  params: Promise<{ locale: string }>;
}) {
  const { locale } = await params;
  const t = await getTranslations({ locale, namespace: 'meta' });
  
  return {
    title: t('home.title'),
    description: t('home.description'),
    openGraph: {
      title: t('home.title'),
      description: t('home.description'),
      url: `https://nakhwa.com.ly/${locale}`,
      siteName: 'Nakhwa',
      locale: locale === 'ar' ? 'ar_LY' : 'en_US',
      type: 'website',
    },
    alternates: {
      canonical: `https://nakhwa.com.ly/${locale}`,
      languages: {
        'ar': 'https://nakhwa.com.ly/ar',
        'en': 'https://nakhwa.com.ly/en',
      },
    },
  };
}
```
